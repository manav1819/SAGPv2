===============================================================================
QUICK START
===============================================================================

1. SETUP IN ROOT LAYOUT (for Toast notifications):
   
   import { ToastProvider } from "@/components/ui";
   
   export default function RootLayout({ children }) {
     return (
       <html>
         <body>
           <ToastProvider>
             {children}
           </ToastProvider>
         </body>
       </html>
     );
   }

2. IMPORT COMPONENTS:
   
   import {
     Button,
     Card,
     Input,
     Badge,
     useToast,
   } from "@/components/ui";

3. USE COMPONENTS:
   
   export function MyComponent() {
     const { addToast } = useToast();
     
     return (
       <Card>
         <Input label="Email" />
         <Button onClick={() => addToast({
           type: "success",
           message: "Done!"
         })}>
           Submit
         </Button>
       </Card>
     );
   }
